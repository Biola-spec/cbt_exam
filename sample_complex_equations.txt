ax² + bx + c = 0
where a=1, b=5, c=6

   x = (-5 ± √(25-24)) / (2)
   x = (-5 ± 1) / 2
   x = -2 or x = -3

   x = (-5 ± 1) / 2
   x = -3 or x = -2

C. x = (-b ± √(b²-4ac)) / (2a)
   x = (5 ± √(25-24)) / (2)
   x = (5 ± 1) / 2
